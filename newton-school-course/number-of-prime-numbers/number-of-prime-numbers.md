## Problem Statement

You are provided with two integers, start and end. Write a JavaScript program to find the number of prime numbers in the range [start, end] (inclusive).

### Input
Inputs will always be positive integers represented by start and end.
10 20

### Output
4 (no of prime numbers in the given range inclusive).

### Example
Example 1:
start = 10;
end = 20;
Number of prime numbers between 10 and 20: 4 (11, 13, 17, 19)

Example 2:
start = 25;
end = 30;
Number of prime numbers between 25 and 30: 1

function isPrime(num) {
  if (num <= 1) return false;
  if (num === 2) return true;
  if (num % 2 === 0) return false;

  for (let i = 3; i * i <= num; i += 2) {
    if (num % i === 0) {
      return false;
    }
  }
  return true;
}

let primeCount = 0;
const low = Math.min(start, end);
const high = Math.max(start, end);

for (let current = low; current <= high; current++) {
  if (isPrime(current)) {
    primeCount++;
  }
}
